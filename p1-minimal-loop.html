5 · 分步实现

第 1 步：模型契约 src/model.ts

先定义整个 agent-lab 唯一依赖的模型接口。它刻意做得很小：一次 generate 收消息列表和工具定义，回文本、工具调用、用量和停止原因。各家 API 的差异都留在适配器里，后面所有项目只认这几个类型。


// 模型层的最小契约：整个 agent-lab 只依赖这几个类型，换厂商只换适配器。
export type ToolCall = { id: string; name: string; args: unknown };

export type Message =
  | { role: 'system' | 'user'; content: string }
  | { role: 'assistant'; content: string; toolCalls: ToolCall[] }
  | { role: 'tool'; toolCallId: string; content: string; isError?: boolean };

export type ToolSpec = { name: string; description: string; inputSchema: Record<string, unknown> };

export type Usage = { inputTokens: number; outputTokens: number; cachedInputTokens: number };

export type StopReason = 'end' | 'tool_use' | 'max_tokens' | 'refusal';

export type GenerateRequest = { messages: Message[]; tools: ToolSpec[]; signal?: AbortSignal };

export type GenerateResult = { content: string; toolCalls: ToolCall[]; usage: Usage; stopReason: StopReason };

export interface Model {
  generate(req: GenerateRequest): Promise<GenerateResult>;
}

export const emptyUsage = (): Usage => ({ inputTokens: 0, outputTokens: 0, cachedInputTokens: 0 });

export const addUsage = (a: Usage, b: Usage): Usage => ({
  inputTokens: a.inputTokens + b.inputTokens,
  outputTokens: a.outputTokens + b.outputTokens,
  cachedInputTokens: a.cachedInputTokens + b.cachedInputTokens,
});
为什么 stopReason 要单列出来

"模型没调用工具"有四种完全不同的含义：正常答完（end）、话说到一半被截断（max_tokens）、拒绝回答（refusal），以及还要继续调工具（tool_use）。把截断和拒答当成正常答案返回，是新手 agent 最常见的静默错误。

第 2 步：假模型 src/fake-model.ts

剧本里每一步可以是固定回复，也可以是函数——函数能读到这一轮的请求，用来模拟"看到某个工具结果后再决定怎么回"。剧本用完了还被调用就报错：多调一次模型本身就是 bug。


import type { GenerateRequest, GenerateResult, Model, ToolCall } from './model';

// 一步脚本：固定回复，或根据收到的请求现算回复（用来断言"模型这一轮看到了什么"）。
export type ScriptStep = Partial<GenerateResult> | ((req: GenerateRequest) => Partial<GenerateResult>);

// 离线测试用的假模型：按脚本依次返回回复，并记下每次收到的请求。
export class FakeModel implements Model {
  readonly requests: GenerateRequest[] = [];
  private cursor = 0;

  constructor(private readonly script: ScriptStep[]) {}

  async generate(req: GenerateRequest): Promise<GenerateResult> {
    req.signal?.throwIfAborted();
    // 存快照：调用方之后还会往同一个数组里 push，不复制的话断言会看到"未来"的消息
    this.requests.push({ messages: structuredClone(req.messages), tools: structuredClone(req.tools) });
    const step = this.script[this.cursor++];
    if (!step) throw new Error(`FakeModel 脚本用完了：第 ${this.cursor} 次调用没有预设回复`);
    const r = typeof step === 'function' ? step(req) : step;
    const toolCalls = r.toolCalls ?? [];
    return {
      content: r.content ?? '',
      toolCalls,
      usage: r.usage ?? { inputTokens: 100, outputTokens: 20, cachedInputTokens: 0 },
      stopReason: r.stopReason ?? (toolCalls.length > 0 ? 'tool_use' : 'end'),
    };
  }
}

let seq = 0;
export const call = (name: string, args: unknown, id = `call_${++seq}`): ToolCall => ({ id, name, args });

第 3 步：工具 src/tools.ts

一个工具 = 名字 + 给模型看的说明 + 参数 schema + 执行函数。用 zod 写 schema 的好处是一份定义两种用途：z.toJSONSchema 转成 JSON Schema 发给模型，执行前再用同一份 schema 校验模型给的参数。


import { resolve, sep } from 'node:path';
import { z } from 'zod';
import type { ToolSpec } from './model';

// 一个工具 = 给模型看的名字和说明 + 参数 schema + 真正执行的代码。
export type Tool<S extends z.ZodType = z.ZodType> = {
  name: string;
  description: string;
  schema: S;
  // 工具已经自带 JSON Schema（例如来自 MCP 服务）时直接交给模型；schema 仍负责本地校验
  jsonSchema?: Record<string, unknown>;
  run(args: z.infer<S>, ctx: { signal?: AbortSignal }): Promise<string>;
};

export const defineTool = <S extends z.ZodType>(tool: Tool<S>): Tool<S> => tool;

// zod → JSON Schema。去掉 $schema 键：有的兼容端点不认识它。
export function toJsonSchema(schema: z.ZodType): Record<string, unknown> {
  const { $schema: _ignored, ...rest } = z.toJSONSchema(schema) as Record<string, unknown>;
  return rest;
}

export const toSpec = (tool: Tool): ToolSpec => ({
  name: tool.name,
  description: tool.description,
  inputSchema: tool.jsonSchema ?? toJsonSchema(tool.schema),
});

export const getTime = defineTool({
  name: 'get_time',
  description: '查询某个时区的当前日期和时间。需要知道"今天几号""现在几点"时调用。',
  schema: z.strictObject({ timezone: z.string().describe('IANA 时区名，例如 Asia/Shanghai、Europe/London') }),
  async run({ timezone }) {
    // 时区写错时 Intl 会抛 RangeError，由循环当成工具错误回灌给模型
    return new Intl.DateTimeFormat('zh-CN', { timeZone: timezone, dateStyle: 'full', timeStyle: 'long' }).format(new Date());
  },
});

export const calculator = defineTool({
  name: 'calculator',
  description: '计算四则运算表达式，支持 + - * / 和括号。涉及任何算术都用它，不要心算。',
  schema: z.strictObject({ expression: z.string().describe('例如 (1200 - 300) * 1.06 / 12') }),
  async run({ expression }) {
    return String(evaluate(expression));
  },
});

// 递归下降求值：只认数字、运算符和括号。绝不用 eval —— 参数来自模型，等于来自不可信输入。
export function evaluate(input: string): number {
  const tokens = input.match(/\d+(?:\.\d+)?|[()+\-*/]|\S/g) ?? [];
  let pos = 0;
  const peek = () => tokens[pos];
  const next = () => tokens[pos++];
  const primary = (): number => {
    const t = next();
    if (t === '(') {
      const v = expr();
      if (next() !== ')') throw new Error('缺少右括号');
      return v;
    }
    if (t === '-') return -primary();
    if (t !== undefined && /^\d/.test(t)) return Number(t);
    throw new Error(`无法识别的符号：${t ?? '（表达式结尾）'}`);
  };
  const term = (): number => {
    let v = primary();
    while (peek() === '*' || peek() === '/') {
      const op = next();
      const r = primary();
      if (op === '/' && r === 0) throw new Error('除数为 0');
      v = op === '*' ? v * r : v / r;
    }
    return v;
  };
  const expr = (): number => {
    let v = term();
    while (peek() === '+' || peek() === '-') v = next() === '+' ? v + term() : v - term();
    return v;
  };
  const value = expr();
  if (pos < tokens.length) throw new Error(`多余的符号：${tokens[pos]}`);
  return value;
}

// 只能读 dir 目录下的笔记：解析成绝对路径后必须仍在目录内，挡住 ../ 和绝对路径。
export function createReadNote(dir: string) {
  const root = resolve(dir);
  return defineTool({
    name: 'read_note',
    description: '读取一篇本地笔记的全文。不知道文件名时，先传 "?" 列出所有笔记。',
    schema: z.strictObject({ name: z.string().describe('笔记文件名，例如 weekly.md；传 "?" 列出全部') }),
    async run({ name }) {
      const all = [...new Bun.Glob('*.md').scanSync(root)].sort();
      if (name === '?') return all.length ? all.join('\n') : '（没有笔记）';
      const path = resolve(root, name);
      if (!path.startsWith(root + sep)) throw new Error('只能读取笔记目录里的文件');
      const file = Bun.file(path);
      if (!(await file.exists())) throw new Error(`没有 ${name}。现有笔记：${all.join('、') || '无'}`);
      return file.text();
    },
  });
}
工具参数等于不可信输入

模型给的参数可能被它读到的内容操纵（第 9 章的提示注入）。所以 calculator 自己写了一个只认数字和运算符的解析器，绝不用 eval；read_note 先把路径解析成绝对路径，再确认它仍在笔记目录里——只检查字符串里有没有 .. 是不够的，绝对路径和符号拼接都能绕过。

第 4 步：循环 src/loop.ts

这就是第 4 章那段伪代码的完整版。逐段对照着读：

  • 步数上限：模型可能在两个工具之间来回绕圈，没有上限就会一直烧钱。超限时抛出的错误带着完整消息记录，方便排查。
  • 每个 tool_call 都要有对应的 tool 消息：模型一次可能调用多个工具；漏掉任何一个的结果，下一次请求会被 API 直接拒绝。
  • 三类错误都回灌、不抛出：工具不存在、参数不合法、执行出错——都变成 isError 的结果交还给模型，让它自己修正。
  • 叫停不是工具错误：用户按了 Ctrl+C，工具里的 fetch 会抛 AbortError；这时要终止整个循环，而不是把它当成"工具失败"回灌。
  • 结果截断：一个工具返回了 50 万字的日志，会把上下文撑爆。截断并告诉模型"已截断，共多少字"。

import { z } from 'zod';
import { addUsage, emptyUsage, type Message, type Model, type ToolCall, type Usage } from './model';
import { type Tool, toSpec } from './tools';

export type AgentEvent =
  | { type: 'model'; step: number; content: string; toolCalls: ToolCall[] }
  | { type: 'tool'; name: string; args: unknown; content: string; isError: boolean };

export type AgentOptions = {
  model: Model;
  tools: Tool[];
  system: string;
  maxSteps?: number;
  maxResultChars?: number;
  signal?: AbortSignal;
  onEvent?: (event: AgentEvent) => void;
};

export type AgentResult = { text: string; messages: Message[]; steps: number; usage: Usage };

// 带上完整消息记录抛出，调用方能看到 agent 在哪一步、因为什么停下。
export class AgentError extends Error {
  constructor(message: string, readonly messages: Message[]) {
    super(message);
  }
}

export async function runAgent(goal: string, opts: AgentOptions): Promise<AgentResult> {
  const maxSteps = opts.maxSteps ?? 10;
  const tools = new Map(opts.tools.map((t) => [t.name, t]));
  const specs = opts.tools.map(toSpec);
  const messages: Message[] = [
    { role: 'system', content: opts.system },
    { role: 'user', content: goal },
  ];
  let usage = emptyUsage();

  for (let step = 1; step <= maxSteps; step++) {
    opts.signal?.throwIfAborted();
    const reply = await opts.model.generate({ messages, tools: specs, signal: opts.signal });
    usage = addUsage(usage, reply.usage);
    messages.push({ role: 'assistant', content: reply.content, toolCalls: reply.toolCalls });
    opts.onEvent?.({ type: 'model', step, content: reply.content, toolCalls: reply.toolCalls });

    if (reply.stopReason === 'refusal') throw new AgentError('模型拒绝了这个请求', messages);
    if (reply.toolCalls.length === 0) {
      if (reply.stopReason === 'max_tokens') throw new AgentError('回答被截断（max_tokens）', messages);
      return { text: reply.content, messages, steps: step, usage };
    }
    // 每个 tool_call 都必须有一条对应的 tool 消息，否则下一次请求会被 API 拒绝
    for (const call of reply.toolCalls) {
      const result = await executeTool(tools, call, opts);
      messages.push({ role: 'tool', toolCallId: call.id, ...result });
      opts.onEvent?.({ type: 'tool', name: call.name, args: call.args, ...result });
    }
  }
  throw new AgentError(`超过最大步数 ${maxSteps} 仍未完成`, messages);
}

// 三类错误都不抛出，而是变成 isError 的结果回灌，让模型自己修正：工具不存在、参数不合法、执行失败。
async function executeTool(tools: Map<string, Tool>, call: ToolCall, opts: AgentOptions) {
  const tool = tools.get(call.name);
  if (!tool) {
    return { isError: true, content: `没有名为 ${call.name} 的工具。可用工具：${[...tools.keys()].join('、')}` };
  }
  const parsed = tool.schema.safeParse(call.args);
  if (!parsed.success) {
    return { isError: true, content: `参数不合法，请修正后重试：\n${z.prettifyError(parsed.error)}` };
  }
  try {
    const out = await tool.run(parsed.data, { signal: opts.signal });
    const max = opts.maxResultChars ?? 8000;
    return { isError: false, content: out.length > max ? `${out.slice(0, max)}\n…（已截断，共 ${out.length} 字）` : out };
  } catch (err) {
    if (opts.signal?.aborted) throw err; // 用户叫停不是工具错误，直接终止
    return { isError: true, content: `工具执行出错：${err instanceof Error ? err.message : String(err)}` };
  }
}

第 5 步：真实模型适配器 src/models/openai-compatible.ts

把内部的消息格式翻译成 Chat Completions 的线上格式。值得注意的几个字段：工具定义在 tools[].function.parameters；模型的工具调用在 message.tool_calls[]，其中 function.arguments 是字符串，不保证是合法 JSON；工具结果用 role: "tool" 加 tool_call_id 回传；finish_reason 为 tool_calls、length、content_filter 时分别对应继续调工具、被截断、被过滤。


import type { GenerateRequest, GenerateResult, Message, Model, StopReason, ToolCall } from '../model';

// OpenAI Chat Completions 兼容接口的适配器：多数模型厂商与本地推理服务都提供这套接口。
// 只用 fetch，不引 SDK，好让你看清线上传的到底是什么。
export type OpenAICompatibleOptions = { baseURL: string; apiKey?: string; model: string; maxTokens?: number };

type WireToolCall = { id: string; type: string; function?: { name: string; arguments: string } };
type WireResponse = {
  choices: { finish_reason: string | null; message: { content: string | null; refusal?: string | null; tool_calls?: WireToolCall[] } }[];
  usage?: { prompt_tokens: number; completion_tokens: number; prompt_tokens_details?: { cached_tokens?: number } };
};

export class OpenAICompatibleModel implements Model {
  constructor(private readonly opts: OpenAICompatibleOptions) {}

  // Bun 会自动读取项目根目录的 .env
  static fromEnv(): OpenAICompatibleModel {
    const model = process.env.LLM_MODEL;
    if (!model) throw new Error('请设置 LLM_MODEL（以及 LLM_BASE_URL、LLM_API_KEY），见 .env.example');
    return new OpenAICompatibleModel({
      baseURL: process.env.LLM_BASE_URL ?? 'https://api.openai.com/v1',
      apiKey: process.env.LLM_API_KEY,
      model,
    });
  }

  async generate({ messages, tools, signal }: GenerateRequest): Promise<GenerateResult> {
    const body = {
      model: this.opts.model,
      messages: messages.map(toWire),
      ...(this.opts.maxTokens ? { max_tokens: this.opts.maxTokens } : {}),
      ...(tools.length > 0
        ? { tools: tools.map((t) => ({ type: 'function', function: { name: t.name, description: t.description, parameters: t.inputSchema } })) }
        : {}),
    };
    const res = await fetch(`${this.opts.baseURL.replace(/\/$/, '')}/chat/completions`, {
      method: 'POST',
      headers: { 'content-type': 'application/json', ...(this.opts.apiKey ? { authorization: `Bearer ${this.opts.apiKey}` } : {}) },
      body: JSON.stringify(body),
      signal,
    });
    if (!res.ok) throw new Error(`模型接口返回 ${res.status}：${(await res.text()).slice(0, 500)}`);
    const data = (await res.json()) as WireResponse;
    const choice = data.choices[0];
    if (!choice) throw new Error('模型接口没有返回 choices');

    const toolCalls: ToolCall[] = (choice.message.tool_calls ?? [])
      .filter((c) => c.type === 'function' && c.function)
      .map((c) => ({ id: c.id, name: c.function!.name, args: parseArgs(c.function!.arguments) }));
    return {
      content: choice.message.content ?? choice.message.refusal ?? '',
      toolCalls,
      usage: {
        inputTokens: data.usage?.prompt_tokens ?? 0,
        outputTokens: data.usage?.completion_tokens ?? 0,
        cachedInputTokens: data.usage?.prompt_tokens_details?.cached_tokens ?? 0,
      },
      stopReason: stopReasonOf(choice.finish_reason, choice.message.refusal),
    };
  }
}

function toWire(m: Message) {
  switch (m.role) {
    case 'system':
    case 'user':
      return { role: m.role, content: m.content };
    case 'assistant':
      return {
        role: 'assistant',
        content: m.content || null,
        ...(m.toolCalls.length > 0
          ? { tool_calls: m.toolCalls.map((c) => ({ id: c.id, type: 'function', function: { name: c.name, arguments: JSON.stringify(c.args) } })) }
          : {}),
      };
    case 'tool':
      return { role: 'tool', tool_call_id: m.toolCallId, content: m.isError ? `[错误] ${m.content}` : m.content };
  }
}

// 模型生成的 arguments 是字符串，不保证是合法 JSON。解析失败就原样交出去，由参数校验报错回灌。
function parseArgs(raw: string): unknown {
  try {
    return JSON.parse(raw);
  } catch {
    return raw;
  }
}

function stopReasonOf(finish: string | null, refusal?: string | null): StopReason {
  if (refusal) return 'refusal';
  if (finish === 'tool_calls') return 'tool_use';
  if (finish === 'length') return 'max_tokens';
  if (finish === 'content_filter') return 'refusal';
  return 'end';
}
arguments 解析失败怎么办

不在适配器里抛错，而是把原始字符串交给循环，让 zod 校验失败、把错误回灌给模型。这样"JSON 写坏了"和"字段写错了"走同一条自我修正的路径，代码也只有一处。

第 6 步：命令行入口 src/main.ts

系统提示里写清三件事：什么时候用工具、出错了怎么办、最多重试几次。打印每一轮的工具调用和结果，你才能看懂它在干什么。


import { type AgentEvent, runAgent } from './loop';

const SYSTEM = `你是一个办公助理。可以查时间、做计算、读用户的本地笔记。

const goal = process.argv.slice(2).join(' ').trim();

const controller = new AbortController();
process.on('SIGINT', () => controller.abort());

const print = (e: AgentEvent) => {
  if (e.type === 'model') {
    if (e.content) console.log(`\x1b[2m[第 ${e.step} 轮] ${e.content}\x1b[0m`);
    for (const c of e.toolCalls) console.log(`\x1b[36m→ ${c.name}(${JSON.stringify(c.args)})\x1b[0m`);
  } else {
    const color = e.isError ? '\x1b[31m' : '\x1b[32m';
    console.log(`${color}← ${e.content.slice(0, 200)}\x1b[0m`);
  }

cp .env.example .env    # 填好 LLM_BASE_URL、LLM_API_KEY、LLM_MODEL

多试几个问题，专门找它的失败：问一个不存在的笔记、给一个错的时区、让它算 1/0、让它读 ../package.json。每次失败时看打印出来的工具调用，判断是提示词、工具说明还是工具实现的问题。

模型一次返回了两个 tool_call：一个查时间、一个读笔记。读笔记的工具抛了异常。正确的处理是？

  1. 立即抛出异常，终止循环
  2. 两个 tool_call 都追加对应的 tool 消息，读笔记那条标为错误，然后继续下一轮
  3. 只追加查时间的结果，跳过失败的那个
  4. 重新请求模型，丢弃这一轮的两个调用
每个 tool_call 都必须有对应的 tool 消息，否则下一次请求会被 API 拒绝（选项 3）。工具失败是常态而不是异常，回灌给模型往往就能自己修正（选项 1 太脆弱）。丢弃这一轮（选项 4）会浪费已经成功的调用，还可能让模型重复做一遍有副作用的操作。

8 · 验收标准

  • bun test 8 个测试全部通过，bun run typecheck 无错误
  • 真实模型能回答需要两个以上工具配合的问题
  • 故意问一个不存在的笔记，能看到工具报错、模型改用 "?" 列出笔记后读对
  • 运行中按 Ctrl+C 能立即停止，不再发出请求
  • 能不看代码，口头讲清 loop.ts 里每个 if 在防什么

9 · 进阶挑战

  1. 并行执行工具：同一轮的多个 tool_call 互不依赖时用 Promise.all 并行跑，但消息顺序必须与调用顺序一致。为它补一个测试。
  2. 流式输出：给适配器加 stream: true，边生成边打印文字；工具调用的参数是分片到达的，要自己拼接。
  3. 预算上限：给 runAgent 加 maxInputTokens，累计用量超限就停，错误里说明花了多少。
  4. 第二家适配器：照着 Model 接口，为另一家原生 API（比如 Anthropic Messages API）写一个适配器，loop.ts 一行都不改。
  5. 工具说明的 A/B：把 read_note 的 description 去掉"不知道文件名时先传 ?"这句，跑十次同一个问题，统计它找对文件的次数。这就是 P6 评测的雏形。

10 · 常见坑

  • FakeModel 不复制请求：循环之后还会往同一个 messages 数组里 push，断言会看到"未来"的消息，测试结果莫名其妙。
  • assistant 消息的 content 传了空字符串：有些兼容端点要求只有工具调用时 content 为 null，适配器里用 m.content || null。
  • 只看 toolCalls.length === 0 判断完成：截断和拒答也是零个工具调用，要先看 stopReason。
  • JSON Schema 里多了 $schema 键：部分端点不认识它会报错，toJsonSchema 里把它去掉。
  • 小模型无视工具：一轮就"答完"了而且答案是编的——换一个支持工具调用的模型，而不是去加长提示词。
本项目要点
  • Agent 的骨架就是"请求模型 → 执行工具 → 结果回灌"的循环，加上步数上限与停止条件。
  • 工具的三类错误都回灌给模型，让它自我修正；用户叫停则立即终止。
  • 一份 zod schema 同时用于"告诉模型"和"校验模型"。
  • 模型的参数是不可信输入：工具要自带边界检查。
  • 循环只依赖 Model 接口，所以能用假模型写出确定性的测试。
